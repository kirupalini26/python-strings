#20 Count vowels and consonants
Count the total vowels and consonants in a given string and print them


s = input()

vowels = 0
consonants = 0

for ch in s.lower():
    if ch.isalpha():
        if ch in "aeiou":
            vowels += 1
        else:
            consonants += 1

print(vowels)
print(consonants)
